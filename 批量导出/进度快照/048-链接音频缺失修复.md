### 缺失音频补追加

修改前：

```js
const copiedItemCount = countConversionTimelineItems(createdTimeline) - itemCountBeforeAppend;
if (copiedItemCount < expectedItemCount) {
  return {
    timeline: createdTimeline,
    reason: `复制源时间线内容失败：目标时间线实际只有 ${Math.max(copiedItemCount, 0)}/${expectedItemCount} 个音视频项目。`
  };
}
```

修改后：

```js
let copiedItemCount = countConversionTimelineItems(createdTimeline) - itemCountBeforeAppend;
let targetRecords = collectConversionTimelineRecords(createdTimeline);
let missingRecords = findMissingTimelineRecords(sourceContent.expectedRecords || [], targetRecords);
let supplementalResult = null;
if (copiedItemCount < expectedItemCount) {
  const missingAudioRecords = selectSupplementalAudioRecords(
    missingRecords,
    targetRecords,
    expectedItemCount - Math.max(copiedItemCount, 0)
  );

  if (missingAudioRecords.length > 0) {
    supplementalResult = appendSupplementalAudioRecords(mediaPool, missingAudioRecords);
    copiedItemCount = countConversionTimelineItems(createdTimeline) - itemCountBeforeAppend;
    targetRecords = collectConversionTimelineRecords(createdTimeline);
    missingRecords = findMissingTimelineRecords(sourceContent.expectedRecords || [], targetRecords);
  }
}
```

### 失败诊断

修改前：

```js
reason: `复制源时间线内容失败：目标时间线实际只有 ${Math.max(copiedItemCount, 0)}/${expectedItemCount} 个音视频项目。`
```

修改后：

```js
reason: buildTimelineCopyFailureReason({
  message: `复制源时间线内容失败：目标时间线实际只有 ${Math.max(copiedItemCount, 0)}/${expectedItemCount} 个音视频项目。`,
  sourceTimeline,
  targetTimeline: createdTimeline,
  sourceContent,
  missingRecords,
  supplementalResult
})
```
