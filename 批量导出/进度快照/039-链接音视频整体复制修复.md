修改前：

```js
collected.push({
  clipInfo: result.clipInfo,

```

修改后：

```js
const linkedAudioRecords = findLinkedAudioRecords(record, audioRecordsByKey, audioRecordsBySignature).filter(
  (audioRecord) => !coveredAudioItemKeys.has(audioRecord.key)
);

appendEntries.push({
  clipInfo: createAllMediaClipInfo(record.clipInfo),
  expectedItemCount: 1 + linkedAudioRecords.length,
  recordFrame: record.recordFrame,
  trackOrder: record.trackOrder,
  trackIndex: record.trackIndex
});
```

修改前：

```js
const appendResult = safeCall(() => mediaPool.AppendToTimeline(sourceContent.clipInfos), null);
const appendedCount = getAppendResultCount(appendResult, sourceContent.clipInfos.length);
if (appendedCount < sourceContent.clipInfos.length) {
  return {
    timeline: createdTimeline,
    reason: `复制源时间线内容失败：只追加 ${appendedCount}/${sourceContent.clipInfos.length} 个剪辑。`
  };
}

const copiedItemCount = countConversionTimelineItems(createdTimeline) - itemCountBeforeAppend;
if (copiedItemCount < sourceContent.clipInfos.length) {
  return {
    timeline: createdTimeline,
    reason: `复制源时间线内容失败：目标时间线实际只有 ${Math.max(copiedItemCount, 0)}/${sourceContent.clipInfos.length} 个音视频项目。`
  };
}
```

修改后：

```js
const appendOperationCount = sourceContent.clipInfos.length;
const expectedItemCount = sourceContent.expectedItemCount || appendOperationCount;
const appendedCount = getAppendResultCount(appendResult, appendOperationCount);
if (appendedCount < appendOperationCount) {
  return {
    timeline: createdTimeline,
    reason: `复制源时间线内容失败：只追加 ${appendedCount}/${appendOperationCount} 次内容。`
  };
}

const copiedItemCount = countConversionTimelineItems(createdTimeline) - itemCountBeforeAppend;
if (copiedItemCount < expectedItemCount) {
  return {
    timeline: createdTimeline,
    reason: `复制源时间线内容失败：目标时间线实际只有 ${Math.max(copiedItemCount, 0)}/${expectedItemCount} 个音视频项目。`
  };
}
```
