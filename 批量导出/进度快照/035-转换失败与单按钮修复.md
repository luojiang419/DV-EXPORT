## 具体修改的代码前后对比

### `plugin-runtime/preload.js`

修改前的主转换路径先复制时间线，再尝试修改复制品帧率，失败后再走 DRT：

```js
const duplicateResult = tryDuplicateTimelineWithFrameRate(sourceTimeline, targetName, targetFrameRate);
if (duplicateResult.timeline && !duplicateResult.reason) {
  createdTimeline = duplicateResult.timeline;
}

if (!createdTimeline) {
  const drtResult = tryImportTimelineFromDrt(...);
}
```

修改后主路径先创建目标帧率空时间线，再导入原时间线内容：

```js
const aafResult = tryImportTimelineViaAaf(
  app,
  project,
  mediaPool,
  sourceTimeline,
  targetFolderResult.folder,
  targetName,
  targetFrameRate
);

if (aafResult.timeline && !aafResult.reason) {
  createdTimeline = aafResult.timeline;
}
```

新增 AAF 导入路径：

```js
createdTimeline = safeCall(() => mediaPool.CreateEmptyTimeline(targetName), null);
createdTimeline.SetSetting("timelineFrameRate", String(targetFrameRate));
project.SetCurrentTimeline(createdTimeline);
createdTimeline.ImportIntoTimeline(aafPath, {
  autoImportSourceClipsIntoMediaPool: false,
  ignoreFileExtensionsWhenMatching: false,
  linkToSourceCameraFiles: false,
  useSizingInfo: true,
  importMultiChannelAudioTracksAsLinkedGroups: true,
  insertAdditionalTracks: true
});
```

修改前 Resolve 导出常量读取：

```js
function getResolveExportConstant(app, key, fallback) {
  return Object.prototype.hasOwnProperty.call(app, key) ? app[key] : fallback;
}
```

修改后直接读取代理/原型属性，并为 DRT/AAF 提供候选导出方式：

```js
function getResolveExportConstant(app, key) {
  const value = safeCall(() => app && app[key], undefined);
  return value === undefined || value === null ? undefined : value;
}

function exportTimelineAsAaf(app, timeline, filePath) {
  return exportTimelineFile(
    app,
    timeline,
    filePath,
    buildTimelineExportCandidates(app, "EXPORT_AAF", ["aaf", "AAF"], ["EXPORT_AAF_NEW", "EXPORT_NONE"])
  );
}
```
